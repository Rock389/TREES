local trunk_blocks = {}
local branches_start_blocks = {}
local branches_blocks = {}
local leaves_blocks = {}
local trunk_leaves_blocks = {}
local leaves_pr = {}
local width_sphere = {}
local outside_width_sphere = {}
local width_circle = {}
local blocks_to_orient = {}
local oriented_blocks_set = {}
local real_leaves_blocks = {}
local real_outside_leaves_blocks = {}
local outside_leaves_blocks = {}
if oriented_blocks ~= nil then
	for _, v in pairs(oriented_blocks) do
		oriented_blocks_set[v] = true
	end
end



local function dsin(angle)
    return math.sin(angle/180*math.pi)
end

local function dcos(angle)
    return math.cos(angle/180*math.pi)
end

local function dtan(angle)
    return dsin(angle)/dcos(angle)
end

local function ceil(num)
    return math.ceil(num)
end

local function shuffle(t)
    for i = #t, 2, -1 do
        local j = math.random(i)
        t[i], t[j] = t[j], t[i]
    end
end

local function to_id(x, y, z)
	return tostring(x) .. " " .. tostring(y) .. " " .. tostring(z)
end

local function num_sign(x)
	if x == 0 then return 0
	elseif x > 0 then return 1
	else return -1 end
end

local function branch_algorithm(x, branch_curve)
	return (-(1 - (dsin(branch_curve)*x)^2)^0.5 + 1) * num_sign(branch_curve)
end

local function generate_trunk()
    local trunk_count = math.random(trunk.min_count, trunk.max_count)
    for j = 1, trunk_count do
        local curr_x, curr_y, curr_z
        local trunk_x, trunk_y, trunk_z
    
		if isSolid(getBlock(x,y,z)) then 
			curr_x, curr_y, curr_z = x, y+1, z
			trunk_x, trunk_y, trunk_z = x, y+1, z
        else
			curr_x, curr_y, curr_z = x, y, z
			trunk_x, trunk_y, trunk_z = x, y, z
		end


        local curr_height = 1
        local trunk_curve_angle = math.random(0, 360)
        local trunk_height = math.random(trunk.min_height, trunk.max_height)
        local trunk_curve = math.random(trunk.min_curve, trunk.max_curve)

        local function one_block_generate()
            local curr_width = (dtan(trunk_height-curr_height)/trunk_height*45) * (trunk.max_width - trunk.min_width) + trunk.min_width
            local temp_block =  trunk.blocks[math.random(1, #trunk.blocks)]
            setBlock(curr_x, curr_y, curr_z, temp_block)
			if oriented_blocks_set[temp_block] then table.insert(blocks_to_orient, {curr_x, curr_y , curr_z, temp_block}) end
            table.insert(trunk_blocks, {curr_x, curr_y, curr_z})
            table.insert(branches_start_blocks, {curr_x, curr_y, curr_z, curr_height, trunk_height})
			leaves_blocks[to_id(curr_x, curr_y, curr_z)] = true

            if curr_width <= 5 then
                table.insert(trunk_leaves_blocks, {curr_x, curr_y, curr_z})
				leaves_pr[to_id(curr_x, curr_y, curr_z)] = curr_height/trunk_height
            end
            
            -- width blocks
            local n = 1
			local r = 1
            local temp_r, temp_n = 0, 0
            while temp_n < curr_width do
                if width_circle[temp_r] == nil then
					width_circle[temp_r] = {}
					for tx = -temp_r,temp_r do
						for ty = -temp_r,temp_r do
							if tx^2 + ty^2 <= temp_r^2 and  tx^2 + ty^2 > (temp_r-1)^2 then
								table.insert(width_circle[temp_r], {tx, ty})
							end
						end
				    end
					shuffle(width_circle[temp_r])
			    end
			    temp_n = temp_n + #width_circle[temp_r]
			    temp_r = temp_r + 1
            end
            
            while n <= curr_width do
                local circle = {}
			    for k, v in pairs(width_circle[r]) do
                    circle[k] = v
                end
				
				if trunk.algorithm == 1 then
					shuffle(circle)
				end
				for ci=1,#circle do
				   local xi, zi = circle[ci][1], circle[ci][2] 
				    if not isSolid(getBlock(curr_x + xi, curr_y, curr_z + zi)) then
			            local temp_block =  trunk.blocks[math.random(1, #trunk.blocks)]
			            setBlock(curr_x + xi, curr_y, curr_z + zi, temp_block)
						if oriented_blocks_set[temp_block] then table.insert(blocks_to_orient, {curr_x+xi, curr_y , curr_z+zi, temp_block}) end
                        table.insert(trunk_blocks, {curr_x + xi, curr_y, curr_z + zi})
						leaves_blocks[to_id(curr_x+xi, curr_y, curr_z+zi)] = true
                        if temp_r - r <= 1 then
                            table.insert(trunk_leaves_blocks, {curr_x + xi, curr_y, curr_z + zi})
							leaves_pr[to_id(curr_x+xi, curr_y, curr_z+zi)] = curr_height/trunk_height
                        end
					end
                	n = n+1
               	 	if n > curr_width then break end
				end
				r = r+1
            end
        end

        for i = 1, trunk_height do
            one_block_generate()
            curr_x = trunk_x + math.floor(curr_height * dsin(i/trunk_height * trunk_curve/2) * dcos(trunk_curve_angle))
            curr_z = trunk_z + math.floor(curr_height * dsin(i/trunk_height * trunk_curve/2) * dsin(trunk_curve_angle))
            curr_y = trunk_y + math.floor(curr_height * dcos(i/trunk_height * trunk_curve/2))
			curr_height = curr_height + 1
        end
    end
end

local function generate_branches()
    local branches_count = math.random(branches.min_count, branches.max_count)
    local branches_start_count = {}
    for i = 1, #branches_start_blocks do
        branches_start_count[i] = 0
    end
  
	local branches_start_min_height = {}
	for i = 1, #branches_start_blocks do
		if branches_start_blocks[i][4] >= branches.min_height then
			table.insert( branches_start_min_height, i)
		end
	end

    for i = 1, branches_count do
	local tt = branches_start_min_height[math.random(1, #branches_start_min_height)]
        branches_start_count[tt] = branches_start_count[tt] + 1
    end

    for i = 1, #branches_start_count do
        while branches_start_count[i] >= 1 do
            local curr_x, curr_y, curr_z = branches_start_blocks[i][1], branches_start_blocks[i][2], branches_start_blocks[i][3]
            local trunk_x, trunk_y, trunk_z, curr_height, trunk_height = branches_start_blocks[i][1], branches_start_blocks[i][2], branches_start_blocks[i][3], branches_start_blocks[i][4], branches_start_blocks[i][5]
 			local branch_length = math.min(math.max((trunk_height-curr_height)/trunk_height + math.random(-0.1, 0.1) ,0), 1) * (branches.max_length - branches.min_length) + branches.min_length
            local branch_curve_angle = math.random(0, 360)
            local branch_curve = math.random(branches.min_curve, branches.max_curve)

            local function one_block_generate(curr_length)
                local curr_width = branches.min_width + dtan((branch_length - curr_length) / branch_length * 45) * (branches.max_width - branches.min_width)
                
                local is_in_trunk_dict = {}
                for _, block in ipairs(trunk_blocks) do
                   is_in_trunk_dict[to_id(block[1], block[2], block[3])] = true
                end
                
                
                if is_in_trunk_dict[to_id(curr_x, curr_y, curr_z)] == nil and getBlock(curr_x, curr_y, curr_z) == blocks.air then
					local temp_block =  branches.blocks[math.random(1, #branches.blocks)]
                    setBlock(curr_x, curr_y, curr_z, temp_block)
					if oriented_blocks_set[temp_block] then table.insert(blocks_to_orient, {curr_x, curr_y , curr_z, temp_block}) end
					if branches["generate_leaves"] == 1 then
						leaves_blocks[to_id(curr_x, curr_y, curr_z)] = true
						leaves_pr[to_id(curr_x, curr_y, curr_z)] = curr_length/branch_length
					end
                    table.insert(branches_blocks, {curr_x, curr_y, curr_z})
                end

                -- width blocks
                local n = 2
                local tries = 0
                while n <= curr_width + 1 and tries < 250 do
                    tries = tries + 1
                    local step = math.floor(ceil(n^(1/3) + 0.001) / 2) + math.floor(tries / 100)
                    local xi, yi, zi
                    
                    local num = math.random(0, 2)
                    if num == 0 then
                        xi = math.random(-step, step)
                        yi = step * (math.random(0, 1) == 1 and -1 or 1)
                        zi = math.random(-step, step)
                    elseif num == 1 then
                        xi = math.random(-step, step)
                        yi = math.random(-step, step)
                        zi = step * (math.random(0, 1) == 1 and -1 or 1)
                    else
                        xi = step * (math.random(0, 1) == 1 and -1 or 1)
                        yi = math.random(-step, step)
                        zi = math.random(-step, step)
                    end
                    
                    if getBlock(curr_x + xi, curr_y + yi, curr_z + zi) == blocks.air and is_in_trunk_dict[to_id(curr_x + xi, curr_y + yi, curr_z + zi)] == nil  then
	                    table.insert(branches_blocks, {curr_x + xi, curr_y + yi, curr_z + zi})
						local temp_block =  branches.blocks[math.random(1, #branches.blocks)]
	                    setBlock(curr_x + xi, curr_y + yi, curr_z + zi, temp_block)
						if oriented_blocks_set[temp_block] then table.insert(blocks_to_orient, {curr_x + xi, curr_y + yi, curr_z + zi, temp_block}) end
						if branches["generate_leaves"] == 1 then
							leaves_blocks[to_id(curr_x+xi, curr_y + yi, curr_z+zi)] = true
							leaves_pr[to_id(curr_x+xi, curr_y + yi, curr_z+zi)] = curr_length/branch_length
	                    end
	                    n = n + 1
                    end
                end
            end

			local curr_length = 1
			local convexity = 1

			if branches.convexity == 2 or branches.convexity == 3 and math.random(1,2) == 1 then convexity = 2 end
            while curr_length <= branch_length do
				local dl = curr_length/branch_length
				phi =  dl * branch_curve/ 2 / 180 * math.pi
				if convexity == 1 then
               		curr_y = trunk_y + math.floor(branch_length* (branch_algorithm(dl, branch_curve)))
				else
					curr_y = trunk_y + math.floor(branch_length* (branch_algorithm(1, branch_curve) - branch_algorithm(1-dl, branch_curve)))
				end
                curr_x = trunk_x + math.floor(curr_length * math.cos(phi) * dcos(branch_curve_angle))
				curr_z = trunk_z + math.floor(curr_length * math.cos(phi) * dsin(branch_curve_angle))

				one_block_generate(curr_length)
				curr_length = curr_length + 1
            end

            branches_start_count[i] = branches_start_count[i] - 1
        end
    end
end

local function generate_leaves()
    local function run_leaves_algorithm(curr_x,curr_y,curr_z,dx,dy,dz,width,real_leaves_blocks,blocks)
				local temp_str = to_id(curr_x + dx, curr_y + dy, curr_z + dz)
				if leaves["algorithm"] == 1 then
	                if math.random(1, 100) <= leaves.density then
						real_leaves_blocks[temp_str] = true
	                    local temp_block =  blocks[math.random(1, #blocks)]
			            setBlock(curr_x + dx, curr_y + dy, curr_z + dz, temp_block)
						if oriented_blocks_set[temp_block] then table.insert(blocks_to_orient, {curr_x + dx, curr_y + dy, curr_z + dz, temp_block}) end
	                end
				elseif leaves["algorithm"] == 2 then
					local cb = 2
					if math.abs(dx) + math.abs(dy) + math.abs(dz) <= 2 then cb = 1/5
					elseif real_leaves_blocks[to_id(curr_x+dx+1,curr_y+dy,curr_z+dz)] or  real_leaves_blocks[to_id(curr_x+dx-1,curr_y+dy,curr_z+dz)] or  real_leaves_blocks[to_id(curr_x+dx,curr_y+dy+1,curr_z+dz)]
						 or  real_leaves_blocks[to_id(curr_x+dx,curr_y+dy-1,curr_z+dz)]  or  real_leaves_blocks[to_id(curr_x+dx,curr_y+dy,curr_z+dz+1)] or  real_leaves_blocks[to_id(curr_x+dx,curr_y+dy,curr_z+dz-1)] then cb = 1/2.5 end
	                if math.random(1, 100) <= (leaves.density/100)^cb*100  then
						real_leaves_blocks[temp_str] = true
						local temp_block =  blocks[math.random(1, #blocks)]
			            setBlock(curr_x + dx, curr_y + dy, curr_z + dz, temp_block)
						if oriented_blocks_set[temp_block] then table.insert(blocks_to_orient, {curr_x + dx, curr_y + dy, curr_z + dz, temp_block}) end
					end
				elseif leaves["algorithm"] == 3 then
					if math.random(1, 100) <= (leaves.density/100) ^ (math.max(1, (dx^2 + dy^2 + dz^2)) / width^2 * (1 - leaves.density/100) * 3) * 100  then
						real_leaves_blocks[temp_str] = true
	                    local temp_block =  blocks[math.random(1, #blocks)]
			            setBlock(curr_x + dx, curr_y + dy, curr_z + dz, temp_block)
						if oriented_blocks_set[temp_block] then table.insert(blocks_to_orient, {curr_x + dx, curr_y + dy, curr_z + dz, temp_block}) end
					end
				end
			end

    if (#leaves.blocks == 1 and leaves.blocks[1] == blocks.air or #leaves.blocks == 0) or leaves.density == 0 then
        return
    end
	
	local leaves_loc = {}    

    if trunk.generate_leaves == 1 then
        for _, block in ipairs(trunk_leaves_blocks) do
            table.insert(leaves_loc, block)
        end
    end

	if branches.generate_leaves == 1 then
        for _, block in ipairs(branches_blocks) do
            table.insert(leaves_loc, block)
        end
    end
    
    local maxdist = 0
    for _, i in ipairs(leaves_loc) do
       local curr_x, curr_y, curr_z = i[1], i[2], i[3]
       maxdist = math.max(maxdist, math.abs(curr_x-x) +math.abs(curr_y-y) + math.abs(curr_z-z))
    end

    for _, i in ipairs(leaves_loc) do
        local curr_x, curr_y, curr_z = i[1], i[2], i[3]
        local dist = math.abs(curr_x-x) + math.abs(curr_y-y) + math.abs(curr_z-z)
        local width = ceil((leaves.max_width-leaves.min_width)*(maxdist-dist)/maxdist) + leaves.min_width
        local fl = width_sphere[width] ~= nil
        
        if not fl then
            local t = {}
			local t2 = {}
            for dx = -width, width do
                for dy = -width, width do
                    for dz = -width, width do
                        if dx^2 + dy^2 + dz^2 <= width^2 then
                            table.insert(t, {dx, dy, dz})
						elseif (dx - 0.5 * num_sign(dx) )^2 + (dy - 0.5 * num_sign(dy))^2 + (dz - 0.5 * num_sign(dz))^2 <= width^2 then
                            table.insert(t2, {dx, dy, dz})
                        end
                    end
                end
            end
            width_sphere[width] = t
			outside_width_sphere[width] = t2
		end

        for _, coord in ipairs(width_sphere[width]) do
			if _%10 > leaves["quality"] then goto continue end
            local dx, dy, dz = coord[1], coord[2], coord[3]
            local temp = {curr_x + dx, curr_y + dy, curr_z + dz}
			local temp_str = to_id(curr_x + dx, curr_y + dy, curr_z + dz)
            
            local is_in_leaves = false
            if leaves_blocks[temp_str] ~= nil then is_in_leaves = true end
			if leaves_pr[to_id(curr_x, curr_y, curr_z)] <  leaves["start_x%"] /100 then goto continue end
			leaves_blocks[temp_str] = true
		         
            if getBlock(curr_x + dx, curr_y + dy, curr_z + dz) == blocks.air and not is_in_leaves and curr_y + dy - y>= leaves.min_height and dy - math.random(-0.5,0.5) >= leaves["start_y"] then
				run_leaves_algorithm(curr_x,curr_y,curr_z,dx,dy,dz,width,real_leaves_blocks,leaves.blocks)
	        end
			::continue::
		end

		if #leaves.outside_blocks > 0 then
			for _, coord in ipairs(outside_width_sphere[width]) do
				if _%10 > leaves["quality"] then goto continue end
	            local dx, dy, dz = coord[1], coord[2], coord[3]
	            local temp = {curr_x + dx, curr_y + dy, curr_z + dz}
				local temp_str = to_id(curr_x + dx, curr_y + dy, curr_z + dz)
	            local is_in_leaves = false
	            if outside_leaves_blocks[temp_str] ~= nil or leaves_blocks[temp_str] ~= nil then is_in_leaves = true end
				if leaves_pr[to_id(curr_x, curr_y, curr_z)] <  leaves["start_x%"] /100 then goto continue end
				outside_leaves_blocks[temp_str] = true
			         
	            if getBlock(curr_x + dx, curr_y + dy, curr_z + dz) == blocks.air and not is_in_leaves and curr_y + dy - y>= leaves.min_height and dy - math.random(-0.5,0.5) >= leaves["start_y"] then
					run_leaves_algorithm(curr_x,curr_y,curr_z,dx,dy,dz,width,real_outside_leaves_blocks,leaves.outside_blocks)
		        end
				::continue::
			end
	    end
	end
end

generate_trunk()
generate_branches()
generate_leaves()

local trunk_blocks_dict = {}
local branches_blocks_dict = {}
if oriented_blocks ~= nil then
	for _, v in pairs(trunk_blocks) do
		trunk_blocks_dict[to_id(v[1], v[2], v[3])] = true
	end
	for _, v in pairs(branches_blocks) do
		branches_blocks_dict[to_id(v[1], v[2], v[3])] = true
	end
end

for _, v in pairs(blocks_to_orient) do
	x, y, z = v[1], v[2], v[3]
	local east = "east=false"
	local west = "west=false"
	local north = "north=false"
	local south = "south=false"
	local up = "up=false"
	local type = "type=bottom"
	local half = "half=bottom"

	if trunk_blocks_dict[to_id(x, y+1, z)] ~= nil or branches_blocks_dict[to_id(x, y+1, z)] ~= nil or real_leaves_blocks[to_id(x, y+1, z)] ~= nil then 
		up = "up=true"
		if trunk_blocks_dict[to_id(x, y-1, z)] ~= nil or branches_blocks_dict[to_id(x, y-1, z)] ~= nil or real_leaves_blocks[to_id(x, y-1, z)] ~= nil then 
			if math.random(0.0, 1.0) > 0.5 then
				type = "type=top"
				half = "half=top"
			end
		else 
			type = "type=top"
			half = "half=top"
		end
	end
	if trunk_blocks_dict[to_id(x+1, y, z)] ~= nil or branches_blocks_dict[to_id(x+1, y, z)] ~= nil or real_leaves_blocks[to_id(x+1, y, z)] ~= nil then
		if getBlockProperty(v[4], "east") == "none" then
			if up == "up=true" then
				east = "east=tall"
			else
				east = "east=low"
			end
		else
			east = "east=true" 
		end
	end
	if trunk_blocks_dict[to_id(x-1, y, z)] ~= nil or branches_blocks_dict[to_id(x-1, y, z)] ~= nil or real_leaves_blocks[to_id(x-1, y, z)] ~= nil then
		if getBlockProperty(v[4], "west") == "none" then
			if up == "up=true" then
				west = "west=tall"
			else
				west = "west=low"
			end
		else
			west = "west=true" 
		end
 	end
	if trunk_blocks_dict[to_id(x, y, z+1)] ~= nil or branches_blocks_dict[to_id(x, y, z+1)] ~= nil or real_leaves_blocks[to_id(x, y, z+1)] ~= nil then
		if getBlockProperty(v[4], "south") == "none" then
			if up == "up=true" then
				south = "south=tall"
			else
				south = "south=low"
			end
		else
			south = "south=true" 
		end
	 end
	if trunk_blocks_dict[to_id(x, y, z-1)] ~= nil or branches_blocks_dict[to_id(x, y, z-1)] ~= nil or real_leaves_blocks[to_id(x, y, z-1)] ~= nil then
		if getBlockProperty(v[4], "north") == "none" then
			if up == "up=true" then
				north = "north=tall"
			else
				north = "north=low"
			end
		else
			north = "north=true" 
		end
	end
	
	setBlock(x, y, z, withBlockProperty(v[4], east, west, north, south, type, half))
end

return nil
