local function dsin(angle)
    return math.sin(angle180math.pi)
end

local function dcos(angle)
    return math.cos(angle180math.pi)
end

local function dtan(angle)
    return dsin(angle)dcos(angle)
end

local function ceil(num)
    return math.ceil(num)
end

local function shuffle(t)
    for i = #t, 2, -1 do
        local j = math.random(i)
        t[i], t[j] = t[j], t[i]
    end
end

local function to_id(x, y, z)
	return tostring(x) ..   .. tostring(y) ..   .. tostring(z)
end

local function num_sign(x)
	if x == 0 then return 0
	elseif x  0 then return 1
	else return -1 end
end

local function rel_sum(x, y, m)
   return (x+y)(1+(xy)(mm))
end

local function set_block(x, y, z, block)
	if oriented_blocks_set[block] then table.insert(blocks_to_orient, {x, y, z, block})
	else setBlock(x, y, z, block) end
end

local function branch_algorithm(x, branch_curve)
	if branches.curve_algorithm == 1 then
		return (-(1 - (dsin(branch_curve)x)^2)^0.5 + 1)  num_sign(branch_curve)
	elseif branches.curve_algorithm == 2 then
		return branch_curve45x
	end
end

local function generate_trunk()
    local trunk_count = math.random(trunk.min_count, trunk.max_count)
    for j = 1, trunk_count do
        local curr_x, curr_y, curr_z
        local trunk_x, trunk_y, trunk_z

		if isSolid(getBlock(x,y,z)) then
			curr_x, curr_y, curr_z = x, y+1, z
			trunk_x, trunk_y, trunk_z = x, y+1, z
        else
			curr_x, curr_y, curr_z = x, y, z
			trunk_x, trunk_y, trunk_z = x, y, z
		end


        local curr_height = 1
        local trunk_curve_angle = math.random(0, 360)
        local trunk_height = math.random(trunk.min_height, trunk.max_height)
        local trunk_curve = math.random(trunk.min_curve, trunk.max_curve)

        local function one_block_generate()
            local curr_width = (dtan(trunk_height-curr_height)trunk_height45)  (trunk.max_width - trunk.min_width) + trunk.min_width
            local temp_block =  trunk.blocks[math.random(1, #trunk.blocks)]
            set_block(curr_x, curr_y, curr_z, temp_block)
            table.insert(trunk_blocks, {curr_x, curr_y, curr_z})
            table.insert(branches_start_blocks, {curr_x, curr_y, curr_z, curr_height, trunk_height})
			leaves_blocks[to_id(curr_x, curr_y, curr_z)] = true

            if curr_width = 5 then
                table.insert(trunk_leaves_blocks, {curr_x, curr_y, curr_z})
				leaves_pr[to_id(curr_x, curr_y, curr_z)] = curr_heighttrunk_height
            end

            -- width blocks
            local n = 1
			local r = 1
            local temp_r, temp_n = 0, 0
            while temp_n  curr_width do
                if width_circle[temp_r] == nil then
					width_circle[temp_r] = {}
					for tx = -temp_r,temp_r do
						for ty = -temp_r,temp_r do
							if tx^2 + ty^2 = temp_r^2 and  tx^2 + ty^2  (temp_r-1)^2 then
								table.insert(width_circle[temp_r], {tx, ty})
							end
						end
				    end
					shuffle(width_circle[temp_r])
			    end
			    temp_n = temp_n + #width_circle[temp_r]
			    temp_r = temp_r + 1
            end

			local last_ci = 0
            local circle = {}
            while n+1 = curr_width do
                circle = {}
			    for k, v in pairs(width_circle[r]) do
                    circle[k] = v
                end

				if trunk.algorithm == 1 then
					shuffle(circle)
				end
				for ci=1,#circle do
				   local xi, zi = circle[ci][1], circle[ci][2]
				    if not isSolid(getBlock(curr_x + xi, curr_y, curr_z + zi)) then
			            local temp_block =  trunk.blocks[math.random(1, #trunk.blocks)]
			            set_block(curr_x + xi, curr_y, curr_z + zi, temp_block)
                        table.insert(trunk_blocks, {curr_x + xi, curr_y, curr_z + zi})
						leaves_blocks[to_id(curr_x+xi, curr_y, curr_z+zi)] = true
                        if temp_r - r = 1 then
                            table.insert(trunk_leaves_blocks, {curr_x + xi, curr_y, curr_z + zi})
							leaves_pr[to_id(curr_x+xi, curr_y, curr_z+zi)] = curr_heighttrunk_height
                        end
					end
					last_ci = ci
               	 	if n+1  curr_width then break end
					n = n+1
				end
				r = r+1
            end

			if #trunk.outside_blocks  0 then
				for ci = last_ci+1,#circle do
					if n = curr_width then
						local xi, zi = circle[ci][1], circle[ci][2]
						if not isSolid(getBlock(curr_x + xi, curr_y, curr_z + zi)) then
							local temp_block =  trunk.outside_blocks[math.random(1, #trunk.outside_blocks)]
							set_block(curr_x + xi, curr_y, curr_z + zi, temp_block)
							table.insert(trunk_blocks, {curr_x + xi, curr_y, curr_z + zi})
							leaves_blocks[to_id(curr_x+xi, curr_y, curr_z+zi)] = true
							n = n + 0.2
						end
					else break end
				end
			end
        end

        for i = 1, trunk_height do
            one_block_generate()
            curr_x = trunk_x + math.floor(curr_height  dsin(itrunk_height  trunk_curve2)  dcos(trunk_curve_angle))
            curr_z = trunk_z + math.floor(curr_height  dsin(itrunk_height  trunk_curve2)  dsin(trunk_curve_angle))
            curr_y = trunk_y + math.floor(curr_height  dcos(itrunk_height  trunk_curve2))
			curr_height = curr_height + 1
        end
    end
end

local function generate_branches()
    local branches_count = math.random(branches.min_count, branches.max_count)
    local branches_start_count = {}
    for i = 1, #branches_start_blocks do
        branches_start_count[i] = 0
    end

	local branches_start_min_height = {}
	for i = 1, #branches_start_blocks do
		if branches_start_blocks[i][4] = branches.min_height then
			table.insert( branches_start_min_height, i)
		end
	end

    for i = 1, branches_count do
	local tt = branches_start_min_height[math.random(1, #branches_start_min_height)]
        branches_start_count[tt] = branches_start_count[tt] + 1
    end

    local is_in_trunk_dict = {}
    for _, block in ipairs(trunk_blocks) do
       is_in_trunk_dict[to_id(block[1], block[2], block[3])] = true
    end

    for i = 1, #branches_start_count do
        while branches_start_count[i] = 1 do
            curr_x, curr_y, curr_z = branches_start_blocks[i][1], branches_start_blocks[i][2], branches_start_blocks[i][3]
            local trunk_x, trunk_y, trunk_z, curr_height, trunk_height = branches_start_blocks[i][1], branches_start_blocks[i][2], branches_start_blocks[i][3], branches_start_blocks[i][4], branches_start_blocks[i][5]
 			local branch_length = math.min(math.max((trunk_height-curr_height)trunk_height + math.random(-0.1, 0.1) ,0), 1)  (branches.max_length - branches.min_length) + branches.min_length
            local branch_curve_angle = math.random(0, 360)
            local branch_curve = math.random(branches.min_curve, branches.max_curve)
           	local curr_branch_main_blocks = {}

            local function one_block_generate(curr_length)
                local curr_width = branches.min_width + dtan((branch_length - curr_length)  branch_length  45)  (branches.max_width - branches.min_width)
				table.insert(curr_branch_main_blocks, {curr_x, curr_y, curr_z, curr_length})
                if is_in_trunk_dict[to_id(curr_x, curr_y, curr_z)] == nil and getBlock(curr_x, curr_y, curr_z) == blocks.air then
    				local temp_block =  branches.blocks[math.random(1, #branches.blocks)]
                    set_block(curr_x, curr_y, curr_z, temp_block)
    				if branches[generate_leaves] == 1 then
    					leaves_blocks[to_id(curr_x, curr_y, curr_z)] = true
    					leaves_pr[to_id(curr_x, curr_y, curr_z)] = curr_lengthbranch_length
    				end
                    table.insert(branches_blocks, {curr_x, curr_y, curr_z})
                end

                local n = 1
    			local r = 1
                local temp_r, temp_n = 0, 0
                while temp_n  curr_width do
                    if width_circle[temp_r] == nil then
    					width_circle[temp_r] = {}
    					for tx = -temp_r,temp_r do
    						for ty = -temp_r,temp_r do
    							if tx^2 + ty^2 = temp_r^2 and  tx^2 + ty^2  (temp_r-1)^2 then
    								table.insert(width_circle[temp_r], {tx, ty})
    							end
    						end
    				    end
    					shuffle(width_circle[temp_r])
    			    end
    			    temp_n = temp_n + #width_circle[temp_r]
    			    temp_r = temp_r + 1
                end

                local last_ci = 0
                local circle = {}
                while n + 1 = curr_width do
                    circle = {}
    			    for k, v in pairs(width_circle[r]) do
                        circle[k] = v
                    end
    				if branches.algorithm == 1 then
						shuffle(circle)
					end
    				for ci=1,#circle do
    			        local xi, zi = circle[ci][1], circle[ci][2]
    			        local xr = math.floor(curr_x + xi  dcos(branch_curve_angle))
    			        local zr = math.floor(curr_z + xi  dsin(branch_curve_angle))
    			        local yr = curr_y + zi
                        if is_in_trunk_dict[to_id(xr, yr, zr)] == nil and getBlock(xr, yr, zr) == blocks.air then
        					local temp_block =  branches.blocks[math.random(1, #branches.blocks)]
                            set_block(xr, yr, zr, temp_block)
        					if branches[generate_leaves] == 1 then
        						leaves_blocks[to_id(xr, yr, zr)] = true
        						leaves_pr[to_id(xr, yr, zr)] = curr_lengthbranch_length
        					end
                            table.insert(branches_blocks, {xr, yr, zr})
                        end
                        last_ci = ci
                        if n+1  curr_width then break end
                        n = n+1
                    end
                    r = r+1
                end

				if #branches.outside_blocks  0 then
					for ci = last_ci+1,#circle do
						if n = curr_width then
							local xi, zi = circle[ci][1], circle[ci][2]
							local xr = math.floor(curr_x + xi  dcos(branch_curve_angle))
							local zr = math.floor(curr_z + xi  dsin(branch_curve_angle))
							local yr = curr_y + zi
							if is_in_trunk_dict[to_id(xr, yr, zr)] == nil and getBlock(xr, yr, zr) == blocks.air then
								local temp_block = branches.outside_blocks[math.random(1, #branches.outside_blocks)]
								set_block(xr, yr, zr, temp_block)
								if branches[generate_leaves] == 1 then
									leaves_blocks[to_id(xr, yr, zr)] = true
									leaves_pr[to_id(xr, yr, zr)] = curr_lengthbranch_length
								end
								table.insert(branches_blocks, {xr, yr, zr})
								n = n + 0.2
							end
						else break end
					end
				end
            end

			local curr_length = 1
			local convexity = 1
			local prev_y = 0
			local phi = 0

			if branches.convexity == 2 or branches.convexity == 3 and math.random(1,2) == 1 then convexity = 2 end
            while curr_length = branch_length do
				local dl = curr_lengthbranch_length
				phi =  dl  branch_curve 2  180  math.pi

				local temp_y = 0
				if convexity == 1 then
               		temp_y = math.floor(branch_length (branch_algorithm(dl, branch_curve)))
				else
					temp_y =  math.floor(branch_length (branch_algorithm(1, branch_curve) - branch_algorithm(1-dl, branch_curve)))
				end
                curr_x = trunk_x + math.floor(curr_length  math.cos(phi)  dcos(branch_curve_angle)  branches.compression)
				curr_z = trunk_z + math.floor(curr_length  math.cos(phi)  dsin(branch_curve_angle)  branches.compression)

                if temp_y  0 then
	                while temp_y  prev_y + 1  do
	                    prev_y = prev_y + 1
	                    curr_y = prev_y + trunk_y
	    				one_block_generate(curr_length)
	    			end
				else
	                while temp_y  prev_y - 1  do
	                    prev_y = prev_y - 1
	                    curr_y = prev_y + trunk_y
	    				one_block_generate(curr_length)
	    			end
				end
    			curr_y = temp_y + trunk_y
				one_block_generate(curr_length)
				curr_length = curr_length + 1
            end

            branches_start_count[i] = branches_start_count[i] - 1
            local branching_count = math.random(branches.min_branching_count, branches.max_branching_count)
            local vertical_count = math.random(branches.min_vertical_branches_count_per_branch, branches.max_vertical_branches_count_per_branch)
    		if curr_branch_main_blocks ~= nil then
    	        for i = 1, branching_count do
    	            local curr_branch_main_block = curr_branch_main_blocks[math.random(1, #curr_branch_main_blocks)]
    	            local curr_branching_length = math.random(branches.min_branching_length_coef(branch_length-curr_branch_main_block[4]), branches.max_branching_length_coef(branch_length-curr_branch_main_block[4]))
    	            curr_x = curr_branch_main_block[1]
    	            curr_y = curr_branch_main_block[2]
    	            curr_z = curr_branch_main_block[3]
    	            prev_y = 0
    	            local branching_curve_angle = branch_curve_angle + math.random(-120, 120)
    	            for dl = 1, curr_branching_length do
    	                curr_x = curr_branch_main_block[1] + math.floor(dl  math.cos(phi)  dcos(branching_curve_angle)  branches.compression)
    					curr_z = curr_branch_main_block[3] + math.floor(dl  math.cos(phi)  dsin(branching_curve_angle)  branches.compression)
    					local temp_y = 0
                        local ddl = (curr_branch_main_block[4]+dl)branch_length
    					if convexity == 1 then
                       		temp_y = math.floor(branch_length  branch_algorithm(ddl, branch_curve))
        				else
        					temp_y = math.floor(branch_length  (branch_algorithm(1, branch_curve) - branch_algorithm(1-ddl, branch_curve)))
        				end
    					curr_y = temp_y + curr_branch_main_block[2]
    					if temp_y  0 then
        	                while temp_y  prev_y + 1  do
        	                    prev_y = prev_y + 1
        	                    curr_y = prev_y + curr_branch_main_block[2]
        	    				one_block_generate(dlcurr_branching_length  (branch_length - curr_branch_main_block[4]) + curr_branch_main_block[4])
        	    			end
        				else
        	                while temp_y  prev_y - 1  do
        	                    prev_y = prev_y - 1
        	                    curr_y = prev_y + curr_branch_main_block[2]
        	    				one_block_generate(dlcurr_branching_length  (branch_length - curr_branch_main_block[4]) + curr_branch_main_block[4])
        	    			end
        				end
    	                one_block_generate(dlcurr_branching_length  (branch_length - curr_branch_main_block[4]) + curr_branch_main_block[4])
    	            end
	            end
	            for i = 1, vertical_count do
	                local curr_branch_main_block = curr_branch_main_blocks[math.random(1, #curr_branch_main_blocks)]
	                curr_x = curr_branch_main_block[1]
	                curr_y = curr_branch_main_block[2]
	                curr_z = curr_branch_main_block[3]
	                local vertical_branch_length = math.random(branches.min_vertical_branches_length, branches.max_vertical_branches_length)
	                for j = 1, vertical_branch_length do
	                    curr_y = curr_y + 1
                        if is_in_trunk_dict[to_id(curr_x, curr_y, curr_z)] == nil and getBlock(curr_x, curr_y, curr_z) == blocks.air then
            				local temp_block =  branches.vertical_branches_blocks[math.random(1, #branches.vertical_branches_blocks)]
                            set_block(curr_x, curr_y, curr_z, temp_block)
            				if branches[generate_leaves] == 1 then
            					leaves_blocks[to_id(curr_x, curr_y, curr_z)] = true
            					leaves_pr[to_id(curr_x, curr_y, curr_z)] = 1.0
            				end
                            table.insert(branches_blocks, {curr_x, curr_y, curr_z})
                        end
	                end
	           end
            end
	    end
	end
end

local function generate_leaves()
    local function run_leaves_algorithm(curr_x,curr_y,curr_z,dx,dy,dz,width,real_leaves_blocks,blocks,density)
				local temp_str = to_id(curr_x + dx, curr_y + dy, curr_z + dz)
				if leaves[algorithm] == 1 then
	                if math.random(1, 100) = density then
						real_leaves_blocks[temp_str] = true
	                    local temp_block =  blocks[math.random(1, #blocks)]
			            set_block(curr_x + dx, curr_y + dy, curr_z + dz, temp_block)
	                end
				elseif leaves[algorithm] == 2 then
					local cb = 2
					if math.abs(dx) + math.abs(dy) + math.abs(dz) = 2 then cb = 15
					elseif real_leaves_blocks[to_id(curr_x+dx+1,curr_y+dy,curr_z+dz)] or  real_leaves_blocks[to_id(curr_x+dx-1,curr_y+dy,curr_z+dz)] or  real_leaves_blocks[to_id(curr_x+dx,curr_y+dy+1,curr_z+dz)]
						 or  real_leaves_blocks[to_id(curr_x+dx,curr_y+dy-1,curr_z+dz)]  or  real_leaves_blocks[to_id(curr_x+dx,curr_y+dy,curr_z+dz+1)] or  real_leaves_blocks[to_id(curr_x+dx,curr_y+dy,curr_z+dz-1)] then cb = 12.5 end
	                if math.random(1, 100) = (density100)^cb100  then
						real_leaves_blocks[temp_str] = true
						local temp_block =  blocks[math.random(1, #blocks)]
			            set_block(curr_x + dx, curr_y + dy, curr_z + dz, temp_block)
					end
				elseif leaves[algorithm] == 3 then
					if math.random(1, 100) = (density100) ^ (math.max(1, (dx^2 + dy^2 + dz^2))  width^2  (1 - density100)  3)  100  then
						real_leaves_blocks[temp_str] = true
	                    local temp_block =  blocks[math.random(1, #blocks)]
			            set_block(curr_x + dx, curr_y + dy, curr_z + dz, temp_block)
					end
				elseif leaves[algorithm] == 4 then
					local cb = 0
					if real_leaves_blocks[to_id(curr_x+dx+1,curr_y+dy,curr_z+dz)] then cb = cb + 1 end
					if real_leaves_blocks[to_id(curr_x+dx-1,curr_y+dy,curr_z+dz)] then cb = cb + 1 end
					if real_leaves_blocks[to_id(curr_x+dx,curr_y+dy+1,curr_z+dz)] then cb = cb + 1 end
					if real_leaves_blocks[to_id(curr_x+dx,curr_y+dy-1,curr_z+dz)] then cb = cb + 1 end
					if real_leaves_blocks[to_id(curr_x+dx,curr_y+dy,curr_z+dz+1)] then cb = cb + 1 end
					if real_leaves_blocks[to_id(curr_x+dx,curr_y+dy,curr_z+dz-1)] then cb = cb + 1 end

                    local c1 = 1
			        local c2 = 1
                    if leaves[compression]  0 then c1 = -leaves[compression]
			        else c2 = leaves[compression] end

					chance = 0
					if cb==2 or cb==3 then chance = 20 end
					chance = rel_sum(chance, (((c1dx)^2 + (c2dy)^2 + (c1dz)^2)  width^2)^0.5  5, 100)
					chance = rel_sum(chance, (density100)^3100, 100)

					if math.random(0, 100) = chance  then
						real_leaves_blocks[temp_str] = true
						local temp_block =  blocks[math.random(1, #blocks)]
			            set_block(curr_x + dx, curr_y + dy, curr_z + dz, temp_block)
					end
				elseif leaves[algorithm] == 5 then
					local cb = 0
					if real_leaves_blocks[to_id(curr_x+dx+1,curr_y+dy,curr_z+dz)] then cb = cb + 1 end
					if real_leaves_blocks[to_id(curr_x+dx-1,curr_y+dy,curr_z+dz)] then cb = cb + 1 end
					if real_leaves_blocks[to_id(curr_x+dx,curr_y+dy+1,curr_z+dz)] then cb = cb + 1 end
					if real_leaves_blocks[to_id(curr_x+dx,curr_y+dy-1,curr_z+dz)] then cb = cb + 1 end
					if real_leaves_blocks[to_id(curr_x+dx,curr_y+dy,curr_z+dz+1)] then cb = cb + 1 end
					if real_leaves_blocks[to_id(curr_x+dx,curr_y+dy,curr_z+dz-1)] then cb = cb + 1 end

                    local c1 = 1
			        local c2 = 1
                    if leaves[compression]  0 then c1 = -leaves[compression]
			        else c2 = leaves[compression] end

					chance = 0
					if cb==2 or cb==3 then chance = 20 end
					chance = rel_sum(chance, 5 - (((c1dx)^2 + (c2dy)^2 + (c1dz)^2)  width^2)^0.5  5, 100)
					chance = rel_sum(chance, (density100)^3100, 100)

					if math.random(0, 100) = chance  then
						real_leaves_blocks[temp_str] = true
						local temp_block =  blocks[math.random(1, #blocks)]
			            set_block(curr_x + dx, curr_y + dy, curr_z + dz, temp_block)
					end
				elseif leaves[algorithm] == 6 then
					local temp_str = to_id(curr_x + dx, curr_y + dy, curr_z + dz)
					if alg6[temp_str] == nil then alg6[temp_str] = {0, 0} end
					alg6[temp_str][1] = alg6[temp_str][1] + (dx^2 + dy^2 + dz^2)width^2 - 0.3
					alg6[temp_str][2] = alg6[temp_str][2] + 1
			    end

			end


	local leaves_loc = {}

    if trunk.generate_leaves == 1 then
        for _, block in ipairs(trunk_leaves_blocks) do
            table.insert(leaves_loc, block)
        end
    end

	if branches.generate_leaves == 1 then
        for _, block in ipairs(branches_blocks) do
            table.insert(leaves_loc, block)
        end
    end

    local maxdist = 0
    for _, i in ipairs(leaves_loc) do
       local curr_x, curr_y, curr_z = i[1], i[2], i[3]
       maxdist = math.max(maxdist, math.abs(curr_x-x) +math.abs(curr_y-y) + math.abs(curr_z-z))
    end

    for _, i in ipairs(leaves_loc) do
        local curr_x, curr_y, curr_z = i[1], i[2], i[3]
        local dist = math.abs(curr_x-x) + math.abs(curr_y-y) + math.abs(curr_z-z)
        local width = ceil((leaves.max_width-leaves.min_width)(maxdist-dist)maxdist) + leaves.min_width
        local fl = width_sphere[width] ~= nil

        if not fl then
            local t = {}
			local t2 = {}
           	local c1 = 1
			local c2 = 1
			if leaves[compression]  0 then c1 = -leaves[compression]
			else c2 = leaves[compression] end
            for dx = math.floor(-widthc1), ceil(widthc1) do
                for dy = math.floor(-widthc2), ceil(widthc2) do
                    for dz = math.floor(-widthc1), ceil(widthc1) do
                        if (c1dx)^2 + (c2dy)^2 + (c1dz)^2 = width^2 then
                            table.insert(t, {dx, dy, dz})
						elseif (c1dx - 0.5  num_sign(dx) )^2 + (c2dy - 0.5  num_sign(dy))^2 + (c1dz - 0.5  num_sign(dz))^2 = width^2 then
                            table.insert(t2, {dx, dy, dz})
                        end
                    end
                end
            end
            width_sphere[width] = t
			outside_width_sphere[width] = t2
		end

        if #leaves.blocks  0 and leaves.density  0 then
            if leaves[algorithm] == 6 then
                alg6 = {}
                indx = {}
            end
            for _, coord in ipairs(width_sphere[width]) do
    			if _%10  leaves[quality] then goto continue end
                local dx, dy, dz = coord[1], coord[2], coord[3]
                local temp = {curr_x + dx, curr_y + dy, curr_z + dz}
    			local temp_str = to_id(curr_x + dx, curr_y + dy, curr_z + dz)

                local is_in_leaves = false
                if leaves_blocks[temp_str] ~= nil then is_in_leaves = true end
    			if leaves_pr[to_id(curr_x, curr_y, curr_z)]   leaves[start_x%] 100 or not (dy - math.random(-0.5,0.5) = leaves[start_y]) or not (dy + math.random(-0.5,0.5) = leaves[end_y]) then goto continue end
    			leaves_blocks[temp_str] = true

    			if leaves[algorithm] == 6 then
    			    indx[temp_str] = temp
    			end

                if getBlock(curr_x + dx, curr_y + dy, curr_z + dz) == blocks.air and not is_in_leaves and curr_y + dy - y= leaves.min_height then
    				run_leaves_algorithm(curr_x,curr_y,curr_z,dx,dy,dz,width,real_leaves_blocks,leaves.blocks,leaves.density)
    	        end
    			continue
    		end

    		if leaves[algorithm] == 6 then
    			for id, vals in pairs(alg6) do
    			    val = vals[1]leaves.density100vals[2]
            		if math.random(0,100) = val100 then
            		    xyz = indx[id]
    					local temp_block =  leaves.blocks[math.random(1, #leaves.blocks)]
    		            set_block(xyz[1], xyz[2], xyz[3], temp_block)
            		end
            	end
            end
        end

		if #leaves.outside_blocks  0 and leaves.outside_density  0 then

		    if leaves[algorithm] == 6 then
                alg6 = {}
                indx = {}
            end

			for _, coord in ipairs(outside_width_sphere[width]) do
				if _%10  leaves[quality] then goto continue end
	            local dx, dy, dz = coord[1], coord[2], coord[3]
	            local temp = {curr_x + dx, curr_y + dy, curr_z + dz}
				local temp_str = to_id(curr_x + dx, curr_y + dy, curr_z + dz)
	            local is_in_leaves = false
	            if outside_leaves_blocks[temp_str] ~= nil or leaves_blocks[temp_str] ~= nil then is_in_leaves = true end
				if leaves_pr[to_id(curr_x, curr_y, curr_z)]   leaves[start_x%] 100 or not (dy - math.random(-0.5,0.5) = leaves[start_y]) or not (dy + math.random(-0.5,0.5) = leaves[end_y]) then goto continue end
				outside_leaves_blocks[temp_str] = true

    			if leaves[algorithm] == 6 then
    			    indx[temp_str] = temp
    			end

	            if getBlock(curr_x + dx, curr_y + dy, curr_z + dz) == blocks.air and not is_in_leaves and curr_y + dy - y= leaves.min_height then
					run_leaves_algorithm(curr_x,curr_y,curr_z,dx,dy,dz,width,real_outside_leaves_blocks,leaves.outside_blocks,leaves.outside_density)
		        end
				continue
			end

    		if leaves[algorithm] == 6 then
    			for id, vals in pairs(alg6) do
    			    val = vals[1]leaves.density100vals[2]
            		if math.random(0,100) = val100 then
            		    xyz = indx[id]
    					local temp_block =  leaves.outside_blocks[math.random(1, #leaves.outside_blocks)]
    		            set_block(xyz[1], xyz[2], xyz[3], temp_block)
            		end
            	end
            end

	    end
	end
end

for _, v in pairs(blocks_to_orient) do
	x, y, z = v[1], v[2], v[3]
	local east = east=false
	local west = west=false
	local north = north=false
	local south = south=false
	local up = up=false
	local type = type=bottom
	local half = half=bottom

	if trunk_blocks_dict[to_id(x, y+1, z)] ~= nil or branches_blocks_dict[to_id(x, y+1, z)] ~= nil or real_leaves_blocks[to_id(x, y+1, z)] ~= nil then
		up = up=true
		if trunk_blocks_dict[to_id(x, y-1, z)] ~= nil or branches_blocks_dict[to_id(x, y-1, z)] ~= nil or real_leaves_blocks[to_id(x, y-1, z)] ~= nil then
			if math.random(0.0, 1.0)  0.5 then
				type = type=top
				half = half=top
			end
		else
			type = type=top
			half = half=top
		end
	end
	if trunk_blocks_dict[to_id(x+1, y, z)] ~= nil or branches_blocks_dict[to_id(x+1, y, z)] ~= nil or real_leaves_blocks[to_id(x+1, y, z)] ~= nil then
		if getBlockProperty(v[4], east) == none then
			if up == up=true then
				east = east=tall
			else
				east = east=low
			end
		else
			east = east=true
		end
	end
	if trunk_blocks_dict[to_id(x-1, y, z)] ~= nil or branches_blocks_dict[to_id(x-1, y, z)] ~= nil or real_leaves_blocks[to_id(x-1, y, z)] ~= nil then
		if getBlockProperty(v[4], west) == none then
			if up == up=true then
				west = west=tall
			else
				west = west=low
			end
		else
			west = west=true
		end
 	end
	if trunk_blocks_dict[to_id(x, y, z+1)] ~= nil or branches_blocks_dict[to_id(x, y, z+1)] ~= nil or real_leaves_blocks[to_id(x, y, z+1)] ~= nil then
		if getBlockProperty(v[4], south) == none then
			if up == up=true then
				south = south=tall
			else
				south = south=low
			end
		else
			south = south=true
		end
	 end
	if trunk_blocks_dict[to_id(x, y, z-1)] ~= nil or branches_blocks_dict[to_id(x, y, z-1)] ~= nil or real_leaves_blocks[to_id(x, y, z-1)] ~= nil then
		if getBlockProperty(v[4], north) == none then
			if up == up=true then
				north = north=tall
			else
				north = north=low
			end
		else
			north = north=true
		end
	end

	setBlock(x, y, z, withBlockProperty(v[4], east, west, north, south, type, half))
end
